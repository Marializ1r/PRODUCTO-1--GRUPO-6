! ============================================================
! SWM2 - Script de configuracion (Equipo 6)
! Pegar en modo privilegiado. Orden recomendado en README.
! ============================================================
enable
config
hostname SWM2

vtp domain EQUIPO6-CORE
vtp mode server
vtp version 2

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,999 priority 8192

interface e0/0
no switchport
ip address 172.26.10.170 255.255.255.252
no shutdown

interface range e0/1-3
switchport trunk encapsulation dot1Q
switchport mode trunk
switchport trunk allowed vlan 10,20,30,40,999
channel-group 1 mode active
no shutdown


interface port-channel 1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,999
 no shutdown

interface e1/0
switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,999
 no shutdown

interface e1/1
switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,999
 no shutdown

interface e1/2
switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,999
 no shutdown
exit

interface vlan 10
 ip address 172.26.8.3 255.255.255.0
 standby 10 ip 172.26.8.1
 standby 10 priority 100
 standby 10 preempt
 no shutdown

interface vlan 20
 ip address 172.26.0.3 255.255.248.0
 standby 20 ip 172.26.0.1
 standby 20 priority 100
 standby 20 preempt
 no shutdown

interface vlan 30
 ip address 172.26.10.67 255.255.255.224
 standby 30 ip 172.26.10.65
 standby 30 priority 100
 standby 30 preempt
 no shutdown

interface vlan 40
 ip address 172.26.10.99 255.255.255.224
 standby 40 ip 172.26.10.97
 standby 40 priority 100
 standby 40 preempt
 no shutdown
exit

interface vlan 10
 ip ospf 10 area 0
interface vlan 20
 ip ospf 10 area 0
interface vlan 30
 ip ospf 10 area 0
interface vlan 40
 ip ospf 10 area 0

router ospf 10
 router-id 10.10.10.2
 network 172.26.10.168 0.0.0.3 area 0
 passive-interface vlan 10
 passive-interface vlan 20
 passive-interface vlan 30
 passive-interface vlan 40
exit

ip access-list standard SOLO-GESTION
 permit 172.26.10.64 0.0.0.31
exit

line vty 0 4
 login local
 transport input ssh
 access-class SOLO-GESTION in
exit

ip access-list extended BLOQUEO-ADMIN
 deny ip 172.26.0.0 0.0.7.255 172.26.10.64 0.0.0.31
 permit ip any any
exit

interface vlan 20
 ip access-group BLOQUEO-ADMIN in
exit


end
wr
