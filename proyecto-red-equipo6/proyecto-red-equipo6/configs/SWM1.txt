! ============================================================
! SWM1 - Script de configuracion (Equipo 6)
! Pegar en modo privilegiado. Orden recomendado en README.

enable
config
hostname SWM1

vtp domain EQUIPO6-CORE
vtp mode server
vtp version 2

vlan 10
 Name DOCENTES
vlan 20
 name ESTUDIANTES
vlan 30
 name GESTION
vlan 40
 name VOZ
vlan 999
 name BLACKHOLE

spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,999 priority 4096

ip routing

interface e0/0
switchport trunk encapsulation dot1Q
no switchport
ip add 172.26.10.166 255.255.255.252
no shut

interface range e0/1-3
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk allowed vlan 10,20,30,40,999
channel-group 1 mode active
no shut

interface port-channel 1
switchport mode trunk
switchport trunk allowed vlan 10,20,30,40,999
no shutdown

interface e1/0
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk allowed vlan 10,20,30,40,999
no shutdown

interface e1/1
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk allowed vlan 10,20,30,40,999
no shutdown

interface e1/2
switchport trunk encapsulation dot1q
switchport mode trunk
switchport trunk allowed vlan 10,20,30,40,999
no shutdown
exit

interface vlan 10
 ip address 172.26.8.2 255.255.255.0
 standby 10 ip 172.26.8.1
 standby 10 priority 110
 standby 10 preempt
 no shutdown

interface vlan 20
 ip address 172.26.0.2 255.255.248.0
 standby 20 ip 172.26.0.1
 standby 20 priority 110
 standby 20 preempt
 no shutdown

interface vlan 30
 ip address 172.26.10.66 255.255.255.224
 standby 30 ip 172.26.10.65
 standby 30 priority 110
 standby 30 preempt
 no shutdown

interface vlan 40
 ip address 172.26.10.98 255.255.255.224
 standby 40 ip 172.26.10.97
 standby 40 priority 110
 standby 40 preempt
 no shutdown
exit

interface vlan 10
 ip ospf 10 area 0
interface vlan 20
 ip ospf 10 area 0
interface vlan 30
 ip ospf 10 area 0
interface vlan 40
 ip ospf 10 area 0

router ospf 10
 router-id 10.10.10.1
 network 172.26.10.164 0.0.0.3 area 0
 passive-interface vlan 10
 passive-interface vlan 20
 passive-interface vlan 30
 passive-interface vlan 40
exit


ip access-list standard SOLO-GESTION
 permit 172.26.10.64 0.0.0.31
exit

line vty 0 4
 login local
 transport input ssh
 access-class SOLO-GESTION in
exit

ip access-list extended BLOQUEO-ADMIN
 deny ip 172.26.0.0 0.0.7.255 172.26.10.64 0.0.0.31
 permit ip any any
exit

interface vlan 20
 ip access-group BLOQUEO-ADMIN in
exit

service dhcp

ip dhcp excluded-address 172.26.8.1 172.26.8.10
ip dhcp excluded-address 172.26.0.1 172.26.0.10
ip dhcp excluded-address 172.26.10.65 172.26.10.70

ip dhcp pool DOCENTES
 network 172.26.8.0 255.255.255.0
 default-router 172.26.8.1
 dns-server 8.8.8.8
exit

ip dhcp pool ESTUDIANTES
 network 172.26.0.0 255.255.248.0
 default-router 172.26.0.1
 dns-server 8.8.8.8
exit

ip dhcp pool GESTION
 network 172.26.10.64 255.255.255.224
 default-router 172.26.10.65
 dns-server 8.8.8.8
exit
